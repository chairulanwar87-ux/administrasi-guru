// Mengambil data profil yang tersimpan di LocalStorage function muatDataTerhubung() { // Ambil data profil (Jika kosong, gunakan default SD NEGERI 1 GUWA LOR) const profil = JSON.parse(localStorage.getItem('profilSekolahGuru')) || { namaSekolah: "SD NEGERI 1 GUWA LOR", tahunAjaran: "2025/2026", semester: "1 (Ganjil)", namaGuru: "Budi Santoso, S.Pd.", nipGuru: "-" }; // Otomatis pasang ke elemen tampilan di halaman cetak/rapor if (document.getElementById('dispSekolah')) { document.getElementById('dispSekolah').innerText = profil.namaSekolah; } if (document.getElementById('dispTA')) { document.getElementById('dispTA').innerText = `: ${profil.tahunAjaran}`; } if (document.getElementById('dispSemester')) { document.getElementById('dispSemester').innerText = `: ${profil.semester}`; } if (document.getElementById('dispWali')) { document.getElementById('dispWali').innerText = `( ${profil.namaGuru} )`; } } // Jalankan otomatis saat halaman selesai dimuat document.addEventListener('DOMContentLoaded', muatDataTerhubung);
